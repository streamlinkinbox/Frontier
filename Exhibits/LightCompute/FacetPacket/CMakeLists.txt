cmake_minimum_required(VERSION 3.20)
project(FacetPacket LANGUAGES CXX)
enable_testing()
add_subdirectory("../FacetAreaOcclusion" "${CMAKE_CURRENT_BINARY_DIR}/native")
add_executable(FacetPacketValidation ../Code/FacetPacketValidation.cpp)
target_compile_features(FacetPacketValidation PRIVATE cxx_std_17)
target_link_libraries(FacetPacketValidation PRIVATE FrontierFacetAreaTransport FrontierFacetAreaReservoir)
if(MSVC)
    target_compile_options(FacetPacketValidation PRIVATE /utf-8 /W4 /fp:strict)
else()
    target_compile_options(FacetPacketValidation PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off)
endif()
add_test(NAME FacetPacketValidation COMMAND FacetPacketValidation --unit)
