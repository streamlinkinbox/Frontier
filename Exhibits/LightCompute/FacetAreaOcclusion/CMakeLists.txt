cmake_minimum_required(VERSION 3.20)
project(FacetAreaOcclusion LANGUAGES CXX)
enable_testing()
add_subdirectory("../FacetAreaTransport" "${CMAKE_CURRENT_BINARY_DIR}/baseline")
add_executable(FacetAreaOcclusionValidation ../Code/FacetAreaOcclusionValidation.cpp)
target_compile_features(FacetAreaOcclusionValidation PRIVATE cxx_std_17)
target_link_libraries(FacetAreaOcclusionValidation PRIVATE FrontierFacetAreaTransport FrontierFacetAreaReservoir)
if(MSVC)
    target_compile_options(FacetAreaOcclusionValidation PRIVATE /utf-8 /W4 /fp:strict)
else()
    target_compile_options(FacetAreaOcclusionValidation PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off)
endif()
add_test(NAME FacetAreaOcclusionValidation COMMAND FacetAreaOcclusionValidation --unit)
