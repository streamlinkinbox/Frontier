cmake_minimum_required(VERSION 3.20)
project(FacetNorm LANGUAGES CXX)
enable_testing()
add_subdirectory("../FacetMixed" "${CMAKE_CURRENT_BINARY_DIR}/mixed")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    ../Code/FacetNormScenes.py ../Code/FacetMixedValidation.cpp)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../Code/FacetNormScenes.py"
    "${CMAKE_CURRENT_BINARY_DIR}/FacetNormScenes.h" COMMAND_ERROR_IS_FATAL ANY)
add_executable(FacetNormValidation ../Code/FacetNormValidation.cpp)
target_include_directories(FacetNormValidation PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" ../Code)
target_compile_features(FacetNormValidation PRIVATE cxx_std_17)
target_link_libraries(FacetNormValidation PRIVATE FrontierFacetAreaTransport FrontierFacetAreaReservoir)
if(MSVC)
    target_compile_options(FacetNormValidation PRIVATE /utf-8 /W4 /fp:strict)
else()
    target_compile_options(FacetNormValidation PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off)
endif()
add_test(NAME FacetNormValidation COMMAND FacetNormValidation --unit)
