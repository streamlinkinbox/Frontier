cmake_minimum_required(VERSION 3.20)
project(FacetMixed LANGUAGES CXX)
enable_testing()
add_subdirectory("../FacetPacket" "${CMAKE_CURRENT_BINARY_DIR}/packet")
add_executable(FacetMixedValidation ../Code/FacetMixedValidation.cpp)
target_compile_features(FacetMixedValidation PRIVATE cxx_std_17)
target_link_libraries(FacetMixedValidation PRIVATE FrontierFacetAreaTransport FrontierFacetAreaReservoir)
if(MSVC)
    target_compile_options(FacetMixedValidation PRIVATE /utf-8 /W4 /fp:strict)
else()
    target_compile_options(FacetMixedValidation PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off)
endif()
add_test(NAME FacetMixedValidation COMMAND FacetMixedValidation --unit)
