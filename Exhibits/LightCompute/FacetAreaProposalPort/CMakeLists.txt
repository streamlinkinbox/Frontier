cmake_minimum_required(VERSION 3.20)
project(FacetAreaProposalPort LANGUAGES CXX)
get_filename_component(Root "../../.." ABSOLUTE)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    ../Code/FacetAreaProposalScenes.py ../Code/TriangleReconnectionValidation.h ../Code/RgbTraceValidation.h
    ../Code/GeometricAreaReservoirValidation.h ../Code/DeformedAreaReservoirValidation.h)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${Root}/Exhibits/LightCompute/Code/FacetAreaProposalScenes.py"
    "${CMAKE_CURRENT_BINARY_DIR}/FacetAreaProposalScenes.h" COMMAND_ERROR_IS_FATAL ANY)
add_subdirectory("${Root}/Frontier/Engine/Rendering" "${CMAKE_CURRENT_BINARY_DIR}/rendering" EXCLUDE_FROM_ALL)
add_executable(FacetAreaProposalValidation ../Code/FacetAreaProposalValidation.cpp)
target_include_directories(FacetAreaProposalValidation PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" ../Code)
target_compile_features(FacetAreaProposalValidation PRIVATE cxx_std_20)
target_link_libraries(FacetAreaProposalValidation PRIVATE FrontierFacetAreaProposals)
add_executable(FacetAreaPortValidation ../Code/FacetAreaPortValidation.cpp)
target_link_libraries(FacetAreaPortValidation PRIVATE FrontierFacetAreaReservoir)
foreach(Target FacetAreaProposalValidation FacetAreaPortValidation)
    if(MSVC)
        target_compile_options(${Target} PRIVATE /utf-8 /W4 /fp:strict)
    else()
        target_compile_options(${Target} PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off)
    endif()
endforeach()
enable_testing()
add_test(NAME FacetAreaProposalValidation COMMAND FacetAreaProposalValidation --unit)
add_test(NAME FacetAreaPortValidation COMMAND FacetAreaPortValidation --unit)
