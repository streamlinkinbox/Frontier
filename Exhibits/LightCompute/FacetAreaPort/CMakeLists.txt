cmake_minimum_required(VERSION 3.20)
project(FacetAreaPort LANGUAGES CXX)
get_filename_component(Root "../../.." ABSOLUTE)
add_subdirectory("${Root}/Frontier/Engine/Rendering" "${CMAKE_CURRENT_BINARY_DIR}/rendering" EXCLUDE_FROM_ALL)
add_executable(FacetAreaPortValidation ../Code/FacetAreaPortValidation.cpp)
target_link_libraries(FacetAreaPortValidation PRIVATE FrontierFacetAreaReservoir)
if(MSVC)
    target_compile_options(FacetAreaPortValidation PRIVATE /utf-8 /W4)
else()
    target_compile_options(FacetAreaPortValidation PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME FacetAreaPortValidation COMMAND FacetAreaPortValidation --unit)
