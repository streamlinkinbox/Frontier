cmake_minimum_required(VERSION 3.20)
project(VehicleLighting LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(Root "../../.." ABSOLUTE)
add_subdirectory("${Root}/Exhibits/LightCompute" inherited)
get_target_property(InheritedSources LightComputeValidation SOURCES)
list(FILTER InheritedSources EXCLUDE REGEX "LightComputeValidation.cpp$")
add_executable(VehicleLightingCapture ../Code/VehicleLightingCapture.cpp ${InheritedSources})
add_executable(VehicleLightingArchive "${Root}/_ClaudeScratch/vehicle-lighting-inputs/VehicleCapture.cpp" ${InheritedSources})
foreach(Target VehicleLightingCapture VehicleLightingArchive)
    target_include_directories(${Target} PRIVATE "${Root}" "${Root}/Frontier" "${VulkanHeaders}")
    target_link_libraries(${Target} PRIVATE FrontierPathReplayDomain FrontierPathReplayVulkan FrontierRadiantFacetDistribution
        FrontierMediumRadiancePayload Threads::Threads ${CMAKE_DL_LIBS})
endforeach()
find_package(Threads REQUIRED)
find_path(VulkanHeaders vulkan/vulkan.h HINTS "${Root}/ExternalPackages/Vulkan-Headers-1.4.321/include" REQUIRED)
target_include_directories(VehicleLightingCapture PRIVATE "${VulkanHeaders}")
target_include_directories(VehicleLightingArchive PRIVATE "${VulkanHeaders}")
